mingw-w64-ucrt-x86_64-ninja              # Sistema de build concurrente de bajo nivel

mingw-w64-ucrt-x86_64-doxygen            # Generador de documentación desde el código fuente
mingw-w64-ucrt-x86_64-clang-tools-extra  # clang-tidy, clang-format y herramientas Clang extra
